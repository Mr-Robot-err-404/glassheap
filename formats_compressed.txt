Compressed 16-bit instruction formats (RVC / Zca)
===========================================================================

FIELD POSITIONS :: these never change, regardless of format
---------------------------------------------------------------------------
  op          ins[1:0]     2 bits   quadrant selector
  funct3      ins[15:13]   3 bits   instruction selector within quadrant
  funct4      ins[15:12]   4 bits
  funct6      ins[15:10]   6 bits
  funct2      ins[6:5]     2 bits
  rd / rs1    ins[11:7]    5 bits   full range x0-x31
  rs2         ins[6:2]     5 bits   full range x0-x31
  rd' / rs1'  ins[9:7]     3 bits   PRIMED, x8-x15 only
  rs2'        ins[4:2]     3 bits   PRIMED, x8-x15 only

primed decode: actual_register = encoded + 8
  000=x8(s0)   001=x9(s1)   010=x10(a0)  011=x11(a1)
  100=x12(a2)  101=x13(a3)  110=x14(a4)  111=x15(a5)

---------------------------------------------------------------------------
WHICH FIELDS EACH FORMAT USES
---------------------------------------------------------------------------
  CR    Register              funct4, rd/rs1, rs2
  CI    Immediate             funct3, rd/rs1, imm
  CSS   Stack-relative Store  funct3, rs2, imm
  CIW   Wide Immediate        funct3, rd', imm
  CL    Load                  funct3, rd', rs1', imm
  CS    Store                 funct3, rs1', rs2', imm
  CA    Arithmetic            funct6, rd'/rs1', funct2, rs2'
  CB    Branch/Arithmetic     funct3, rd'/rs1', offset
  CJ    Jump                  funct3, jump target

===========================================================================
INSTRUCTION REFERENCE
===========================================================================
format gives register positions (see FIELD POSITIONS above).
imm layout is per-instruction. sx = sign-extend from N bits, zx = zero-extend.

---------------------------------------------------------------------------
QUADRANT 0   op=00
---------------------------------------------------------------------------
funct3  name         fmt   imm layout                                    -> 32-bit

000     C.ADDI4SPN   CIW   ins[10:7]<<6 | ins[12:11]<<4 | ins[5]<<3 | ins[6]<<2   zx
                           -> ADDI rd', x2, imm

010     C.LW         CL    ins[5]<<6 | ins[12:10]<<3 | ins[6]<<2         zx
                           -> LW rd', imm(rs1')

011     C.LD         CL    ins[6:5]<<6 | ins[12:10]<<3                   zx
                           -> LD rd', imm(rs1')

110     C.SW         CS    ins[5]<<6 | ins[12:10]<<3 | ins[6]<<2         zx
                           -> SW rs2', imm(rs1')

111     C.SD         CS    ins[6:5]<<6 | ins[12:10]<<3                   zx
                           -> SD rs2', imm(rs1')

---------------------------------------------------------------------------
QUADRANT 1   op=01
---------------------------------------------------------------------------
funct3  name         fmt   imm layout                                    -> 32-bit

000     C.NOP        -     rd=0, imm=0
                           -> ADDI x0, x0, 0

000     C.ADDI       CI    ins[12]<<5 | ins[6:2]                         sx 6
                           -> ADDI rd, rd, imm

001     C.ADDIW      CI    ins[12]<<5 | ins[6:2]                         sx 6
                           -> ADDIW rd, rd, imm                          (RV64)

010     C.LI         CI    ins[12]<<5 | ins[6:2]                         sx 6
                           -> ADDI rd, x0, imm

011     C.ADDI16SP   CI    ins[12]<<9 | ins[4:3]<<7 | ins[5]<<6 |
                           ins[2]<<5 | ins[6]<<4                         sx 10
                           -> ADDI x2, x2, imm                           (rd == x2)

011     C.LUI        CI    ins[12]<<17 | ins[6:2]<<12                    sx 18
                           -> LUI rd, imm                                (rd != x2)

100/00  C.SRLI       CB    ins[12]<<5 | ins[6:2]                         zx (shamt)
                           -> SRLI rd', rd', shamt

100/01  C.SRAI       CB    ins[12]<<5 | ins[6:2]                         zx (shamt)
                           -> SRAI rd', rd', shamt

100/10  C.ANDI       CB    ins[12]<<5 | ins[6:2]                         sx 6
                           -> ANDI rd', rd', imm

100     C.SUB        CA    funct6=100011 funct2=00   -> SUB  rd', rd', rs2'
100     C.XOR        CA    funct6=100011 funct2=01   -> XOR  rd', rd', rs2'
100     C.OR         CA    funct6=100011 funct2=10   -> OR   rd', rd', rs2'
100     C.AND        CA    funct6=100011 funct2=11   -> AND  rd', rd', rs2'
100     C.SUBW       CA    funct6=100111 funct2=00   -> SUBW rd', rd', rs2'  (RV64)
100     C.ADDW       CA    funct6=100111 funct2=01   -> ADDW rd', rd', rs2'  (RV64)

101     C.J          CJ    ins[12]<<11 | ins[8]<<10 | ins[10:9]<<8 |
                           ins[6]<<7 | ins[7]<<6 | ins[2]<<5 |
                           ins[11]<<4 | ins[5:3]<<1                      sx 12
                           -> JAL x0, imm

110     C.BEQZ       CB    ins[12]<<8 | ins[6:5]<<6 | ins[2]<<5 |
                           ins[11:10]<<3 | ins[4:3]<<1                   sx 9
                           -> BEQ rs1', x0, imm

111     C.BNEZ       CB    ins[12]<<8 | ins[6:5]<<6 | ins[2]<<5 |
                           ins[11:10]<<3 | ins[4:3]<<1                   sx 9
                           -> BNE rs1', x0, imm

---------------------------------------------------------------------------
QUADRANT 2   op=10
---------------------------------------------------------------------------
funct3  name         fmt   imm layout                                    -> 32-bit

000     C.SLLI       CI    ins[12]<<5 | ins[6:2]                         zx (shamt)
                           -> SLLI rd, rd, shamt

010     C.LWSP       CI    ins[3:2]<<6 | ins[12]<<5 | ins[6:4]<<2        zx
                           -> LW rd, imm(x2)

011     C.LDSP       CI    ins[4:2]<<6 | ins[12]<<5 | ins[6:5]<<3        zx
                           -> LD rd, imm(x2)                             (RV64)

100     C.JR         CR    ins[12]=0  rs2=0          -> JALR x0, rs1, 0
100     C.MV         CR    ins[12]=0  rs2!=0         -> ADD  rd, x0, rs2
100     C.EBREAK     CR    ins[12]=1  rd=0  rs2=0    -> EBREAK
100     C.JALR       CR    ins[12]=1  rs2=0          -> JALR x1, rs1, 0
100     C.ADD        CR    ins[12]=1  rs2!=0         -> ADD  rd, rd, rs2

110     C.SWSP       CSS   ins[8:7]<<6 | ins[12:9]<<2                    zx
                           -> SW rs2, imm(x2)

111     C.SDSP       CSS   ins[9:7]<<6 | ins[12:10]<<3                   zx
                           -> SD rs2, imm(x2)                            (RV64)
